package main

// Documentation-only observation probe: runs actual public Go code with
// synthetic offline PARAM frames. It is NOT a Delphi executable oracle.
// A PASS means the observed implementation matches the stated baseline,
// not that it has achieved source-derived trigger-prefix parity.
import (
    "encoding/json"
    "os"
    "path/filepath"
    "strings"
    "testing"
    "time"
)

func TestAtlasV19ActualParamOrder(t *testing.T) {
    block := func(addr uint16, bits uint32) RegBlock {
        return RegBlock{Addr:addr, Words:2, Payload:sourceRawTestWords(bits)}
    }
    seed := block(0x2002,0x3f800000)
    two := block(0x2002,0x40000000)
    three := block(0x2002,0x40400000)
    trigger := block(0x3100,0)
    for _, tc := range []struct{
        name string
        blocks []RegBlock
        inactive, frozen, invalid bool
        wantRows int
        wantRow, wantFinal, sourcePrefix string
    }{
        {"trigger_only", []RegBlock{trigger},false,false,false,1,"0000803f","0000803f","0000803f"},
        {"before", []RegBlock{two,trigger},false,false,false,1,"00000040","00000040","00000040"},
        {"after", []RegBlock{trigger,two},false,false,false,1,"00000040","00000040","0000803f"},
        {"both_sides", []RegBlock{two,trigger,three},false,false,false,1,"00004040","00004040","00000040"},
        {"duplicate", []RegBlock{trigger,two,trigger},false,false,true,0,"","00000040","grammar_not_attested"},
        {"freeze", []RegBlock{trigger,two},false,true,false,0,"","0000803f","no_row"},
        {"inactive", []RegBlock{trigger,two},true,false,false,0,"","00000040","no_row"},
        {"no_trigger", []RegBlock{two},false,false,false,0,"","00000040","no_row"},
    } {
        t.Run(tc.name,func(t *testing.T){
            m,c:=markFixture(t)
            defer m.CloseFiles()
            m.sourceLiveValuesWriter=newSourceLiveValuesWriter(m.dir)
            m.sourceOriginalRXGate=newSourceOriginalReceiveGate()
            var ch [10]bool; ch[0]=true
            var spec [43]bool; spec[0]=true
            var values [73]bool
            if !tc.inactive { sourceSelectedMark(t,c,m,1,"values_begin",ch,spec,values) }
            at:=time.Date(2026,11,24,4,0,0,0,time.UTC)
            if err:=m.ProcessChunk(makeParam([]RegBlock{seed}),at);err!=nil{t.Fatal(err)}
            m.sourceOriginalRXGate.frozen=tc.frozen
            packet:=makeParam(tc.blocks)
            if _,err:=decodeBlocks(packet);err!=nil{t.Fatalf("Go framing rejects test vector: %v",err)}
            if err:=m.ProcessChunk(packet,at.Add(time.Second));err!=nil{t.Fatal(err)}
            invalid:=m.summary.InvalidFrames>0
            if invalid!=tc.invalid{t.Errorf("invalid got %v want %v",invalid,tc.invalid)}
            final:=m.sourceLiveRawDWORD(0x2002)
            if final==nil || *final!=tc.wantFinal{t.Fatalf("unexpected final register: %v",final)}
            report,_,err:=m.Finalize("STOPPED_BY_USER");if err!=nil{t.Fatal(err)}
            if int(report.SourceLiveValuesEventRecords)!=tc.wantRows{t.Fatalf("row count %d want %d",report.SourceLiveValuesEventRecords,tc.wantRows)}
            got:=""
            if tc.wantRows>0 {
                raw,err:=os.ReadFile(filepath.Join(m.dir,"SOURCE_W01_VALUES_3100_RECEIVED.jsonl"));if err!=nil{t.Fatal(err)}
                for _,line:=range strings.Split(strings.TrimSpace(string(raw)),"\n"){
                    var ev sourceLiveValuesEvent
                    if err=json.Unmarshal([]byte(line),&ev);err!=nil{t.Fatal(err)}
                    if ev.RawTickEvidence==nil || len(ev.RawTickEvidence.OrderedFields)!=1 || ev.RawTickEvidence.OrderedFields[0].RawLE==nil{t.Fatal("missing raw field evidence")}
                    got=*ev.RawTickEvidence.OrderedFields[0].RawLE
                }
                if got!=tc.wantRow{t.Errorf("row %q want %q",got,tc.wantRow)}
            }
            t.Logf("OBSERVED name=%s rows=%d row=%s final=%s invalid=%v source_derived_prefix=%s",tc.name,report.SourceLiveValuesEventRecords,got,*final,invalid,tc.sourcePrefix)
        })
    }
}

func TestAtlasV19ActualTachFreshness(t *testing.T) {
    for _, after:=range []bool{false,true} {
        name:="before";if after{name="after"}
        t.Run(name,func(t *testing.T){
            m,c:=markFixture(t);defer m.CloseFiles()
            m.sourceLiveValuesWriter=newSourceLiveValuesWriter(m.dir)
            var ch [10]bool;ch[8]=true
            var spec [43]bool
            var values [73]bool;values[35]=true
            sourceSelectedMark(t,c,m,1,"values_begin",ch,spec,values)
            at:=time.Date(2026,11,24,4,0,0,0,time.UTC)
            tach:=RegBlock{Addr:0x30fe,Words:2,Payload:sourceRawTestWords(0x12345678)}
            trigger:=RegBlock{Addr:0x3100,Words:2,Payload:sourceRawTestWords(0)}
            blocks:=[]RegBlock{tach,trigger};if after{blocks=[]RegBlock{trigger,tach}}
            if err:=m.ProcessChunk(makeParam(blocks),at);err!=nil{t.Fatal(err)}
            if _,_,err:=m.Finalize("STOPPED_BY_USER");err!=nil{t.Fatal(err)}
            raw,err:=os.ReadFile(filepath.Join(m.dir,"SOURCE_W01_VALUES_3100_RECEIVED.jsonl"));if err!=nil{t.Fatal(err)}
            var ev sourceLiveValuesEvent
            if err=json.Unmarshal(raw,&ev);err!=nil{t.Fatal(err)}
            if ev.RawTickEvidence==nil || len(ev.RawTickEvidence.OrderedFields)!=1{t.Fatal("missing tach evidence")}
            field:=ev.RawTickEvidence.OrderedFields[0]
            if field.RawLE==nil || *field.RawLE!="78563412" || !field.SourceWordValid{t.Fatalf("unexpected current frame freshness: %+v",field)}
            t.Logf("OBSERVED tach_position=%s raw=%s valid=%v (source-prefix freshness after trigger is not available at trigger)",name,*field.RawLE,field.SourceWordValid)
        })
    }
}
